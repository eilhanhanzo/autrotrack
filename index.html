// URL of your published Google Sheet CSV const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vRwJdCYn7XQAignh4mbDT_T-xlfer0SJcJQbPq-UgGET6vLAnGE4JAVLe5T9hq9F2gbHpEVcpuHOBVB/pub?gid=240399308&single=true&output=csv"; let map; let markers = []; let latestTimestampElement; window.onload = function () { initMap(); latestTimestampElement = document.getElementById("timestamp"); fetchAndPlot(); setInterval(fetchAndPlot, 5000); // refresh every 5 seconds }; function initMap() { map = L.map("map").setView([5.70, 115.94], 14); L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", { maxZoom: 19, }).addTo(map); } function fetchAndPlot() { fetch(CSV_URL) .then(response => response.text()) .then(csvText => { const rows = csvText.trim().split("\n").slice(1); // skip header const points = []; rows.forEach(row => { const cols = row.split(","); let lat = parseFloat(cols[0]); let lng = parseFloat(cols[1]); let timestamp = cols[2] || ""; if (!isNaN(lat) && !isNaN(lng)) { points.push({ lat, lng, timestamp }); } }); plotDots(points); }) .catch(err => console.error("CSV Load Error:", err)); } function plotDots(points) { // remove old markers markers.forEach(m => map.removeLayer(m)); markers = []; if (points.length === 0) return; let last = points[points.length - 1]; // show timestamp of latest coordinate latestTimestampElement.innerText = "Latest timestamp: " + last.timestamp; points.forEach(p => { let marker = L.circleMarker([p.lat, p.lng], { radius: 5, color: "red", fillColor: "red", fillOpacity: 1 }).addTo(map); marker.bindPopup("Time: " + p.timestamp); markers.push(marker); }); map.setView([last.lat, last.lng]); }
